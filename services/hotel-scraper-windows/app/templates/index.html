Database Verification Queries

Run against the PostgreSQL hotel_scraper database via the Cloud SQL Auth Proxy (read-only checks recommended).

-- Total hotels count
SELECT count(*) FROM hotels;

-- Sources breakdown (Places-only vs OSM-only vs Merged)
SELECT sources, count(*)
FROM hotels
GROUP BY sources
ORDER BY count(*) DESC;

-- ZIP queue completion breakdown
SELECT
  count(*) AS total_zips,
  count(*) FILTER (WHERE places_status = 'done') AS places_done,
  count(*) FILTER (WHERE places_status = 'error') AS places_failed
FROM zips;

-- List currently failed ZIP codes
SELECT zip, city, state, dist_km_from_kirkland, places_status, last_error
FROM zips
WHERE places_status = 'error';
